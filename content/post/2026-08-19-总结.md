### 统一启动函数
```cpp
// 1. 注册停止回调
Utils::Exit::RegisterStopCallback([]() {
    // 清理本服务资源（Stop、Join、清空等）
    if (g_server) g_server->Stop();
    ShutDownMySQL();
});

// 2. 注册统一信号处理
std::signal(SIGINT, Utils::Exit::Onsignal);
#ifdef _WIN32
SetConsoleCtrlHandler(Utils::Exit::ConsoleCtrlHandler, TRUE);
#endif

// 3. 服务主循环
while (Utils::Exit::running.load()) {
    // ... 处理消息，收到"close"时：
    // Utils::Exit::RequestExit(); break;
}

// 4. 主线程等待退出事件（阻塞，不占CPU）
Utils::Exit::WaitForExit();

// 5. 退出后的残留清理（幂等，可重复）
g_server.reset();
```
