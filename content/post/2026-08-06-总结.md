# RUST

## 切片

- 不持有所有权的数据类型
- 指向字符串一部分的引用

```rust
let hello=&s[0..5];
```

- 相当于c++的strsub
- 前面不写默认为0，后面不写默认到结尾
- 字符串字面值被直接存储在二进制程序中
- 如果切片，则字符串不可变
- 使用&str作为参数类型，就可以同时接收string和&str类型了：这样可以使得api更加通用，也不会有功能损失

## struct

- 结构体
- 自定义数据类型

### 定义

```rust
struct User{
    username:String,
    email:String
    ……
}
```

- 和json类似

### 取值

- 用点标法

### 赋值

```rust
user1.email=String::from(".....@....com")
```

- 实例有一个可变则整个字段可变

### 更新语法

```rust
let user2=User{
    user1
}
```

### tup struct

- 定义类似tuple的struct
- 整体有名，元素无名

```rust
struct tup_struct_name(type_name,type_name);
```

- 即使类型相同则也是两个结构体

### Unit-Like Struct

- 空结构体

### 所有权

- struct拥有其所有的数据
- 当String不是&str时，只要struct实例有效，则字段数据也是有效的
- strcut也可以引用，但是需要使用生命周期

```cpp
#pragma once
#include <iostream>
#include <string>
#include <boost/asio.hpp>
#include <memory>
#include <cstring>
#include <vector>
#include <utility>
#include <deque>
#include <atomic>

namespace Messaging{
    constexpr int MAX_LEN=1024*1024;
    constexpr int HEAD_LEN=4;
    constexpr int HEAD_ID_LEN=4;

    //消息体的缓存类
    class MsgNode{
        public:
        //构造消息体
        MsgNode(int);
        //析构消息体
        virtual ~MsgNode();
        //清空缓存
        void Clear();
        // 获取缓冲区指针
        char* GetBuf() const { return buf; }
        // 获取缓冲区总长度
        int GetTotalLen() const { return total_len; }
        // 获取当前读取位置
        int GetCurLen() const { return cur_len; }
        // 设置当前读取位置
        void SetCurLen(int len) { cur_len = len;}

        //禁用拷贝构造函数
        MsgNode(const MsgNode&) = delete;
        //禁用拷贝赋值函数
        MsgNode& operator=(const MsgNode&) = delete;
        protected:
        char* buf;//消息体缓存
        int total_len;//缓存数据大小
        int cur_len;//当前读取到的位置
    };

    //读取类
    class RecvNode:public MsgNode{
        public:
        RecvNode(int ,int );
        //获取ID
        int GetID()const{return msg_id;}
        private:
        //消息ID
        int msg_id;
    };

    class SendNode:public MsgNode{
        public:
        SendNode(int,int);
        //获取ID
        int GetID()const{return msg_id;}
        private:
        //消息ID
        int msg_id;
    };

    //处理连接
    class Session:public std::enable_shared_from_this<Session>//具体读写措施类
    {
        public:
        //构造函数
        Session(boost::asio::io_context& ioc,boost::asio::ip::tcp::socket socket)
        :ioc(ioc),stop(false),sock(std::move(socket)){}
        //外部启动函数
        void Start();
        //停止关闭函数
        void Stop();
        private:
        //异步读函数
        void ReadHead();
        //读消息体函数
        void ReadBody(int,int);
                //异步写函数
        void Write(int ,const std::string&);
        //执行发送队列
        void DoSend();

        //工作函数
        void Work(int,std::string&);

                //io_context引用（用于跨线程投递停止任务）
        boost::asio::io_context& ioc;
        //保存是否退出状态（原子变量，防止跨线程数据竞争）
        std::atomic<bool> stop;
        //保存socket
        boost::asio::ip::tcp::socket sock;
        //读节点
        std::shared_ptr<RecvNode> recv_node;
        //写节点
        std::shared_ptr<SendNode> send_node;
        //发送队列
        std::deque<std::shared_ptr<SendNode>> send_queue;
        //是否正在发送
        bool is_sending = false;
        //头部长度
        static const int HEADER_LENGTH=HEAD_ID_LEN+HEAD_LEN;
    };

    //服务器主程序，分配Session
    class Server
    {
    public:
        //构造Server
        Server(boost::asio::io_context&,unsigned short);

        //Server停止函数（可在任意线程调用，内部会投递到io_context线程执行）
        void Stop();
        //获取io_context
        boost::asio::io_context& GetIoc() const { return ioc; }
        private:
        //分配Session
        void To_Session();

        //io_context引用（用于跨线程投递停止任务）
        boost::asio::io_context& ioc;
        //连接的accept
        boost::asio::ip::tcp::acceptor acceptor;
        //管理连接socket
        std::shared_ptr<boost::asio::ip::tcp::socket> sock;
        //管理Session的连接
        std::vector<std::shared_ptr<Session>> sessions;
    };

    //启动命令
    void start(unsigned short port);
}
```

```cpp
#include "messaging.h"

namespace Messaging{
    //构造函数
    MsgNode::MsgNode(int max_len){
        //计算出缓存空间
        total_len=max_len;
        //设定指针
        cur_len=0;
        //申请缓存
        buf=new char[total_len+1];
        //给最后一个空间为'\0'避免超出空间
        buf[total_len]='\0';
    }
    //析构函数
    MsgNode::~MsgNode(){
        //直接删除动态函数
        delete[] buf;
    }
    //清空缓存
    void MsgNode::Clear(){
        //给缓存赋值为'\0'
        ::memset(buf,'\0',total_len);
        //移动指针复位
        cur_len=0;
    }

    //接收节点
    RecvNode::RecvNode(int max_len,int _msg_id):MsgNode(max_len)
    {
        //获得ID
        msg_id=_msg_id;
    }

    //发送节点
    SendNode::SendNode(int max_len,int _msg_id):MsgNode(max_len)
    {
        //发送ID
        msg_id=_msg_id;
    }

    //Session调用的启动函数
    void Session::Start(){
        ReadHead();
    }

        //停止发送函数（线程安全：可在任意线程调用，内部会投递到io_context线程执行关闭检查）
        void Session::Stop(){
            //原子写停止标志（任意线程安全）
            stop=true;
            //捕获自身，防止投递的任务执行时对象已被销毁
            auto self=shared_from_this();
            //投递关闭检查到io_context线程，避免跨线程访问socket和队列
            boost::asio::post(ioc,[self,this](){
                //若发送队列为空且没有正在发送，则立即关闭连接
                if(send_queue.empty() && !is_sending){
                    boost::system::error_code ec;
                    sock.close(ec);
                }
            });
        }

    //读数据
    void Session::ReadHead()
    {
        //先读取到头部的消息ID和消息长度
        recv_node=std::make_shared<RecvNode>(HEADER_LENGTH,0);
        //初始化缓存
        recv_node->Clear();
        //保活
        auto self=shared_from_this();
        //读取数据
        boost::asio::async_read(
            sock,boost::asio::buffer(recv_node->GetBuf(),recv_node->GetTotalLen()),
            [this,self](const boost::system::error_code& ec,
                        std::size_t)
                    {
                        //判断是否有错误
                        if(ec){
                            //有，输出错误并返回
                            std::cout<<"error in run , value is : "<< ec.message()<<std::endl;
                            return;
                        }
                        //判断是否为关闭状态
                        if(stop){
                            //提示仍收到消息
                            std::cout<<"closing,but recive message"<<std::endl;
                            //回发送提示（入队，发送完毕后再关闭）
                            Write(0,"server is closing,please not connect");
                            return;
                        }
                        //获取读缓存指针
                        char* header =recv_node->GetBuf();
                        //获取信息ID和信息长度
                        int msg_id=0;
                        int msg_len=0;

                        //将信息ID和信息长度，读出缓存
                        std::memcpy(&msg_id,header,HEAD_ID_LEN);
                        std::memcpy(&msg_len,header+HEAD_ID_LEN,HEAD_LEN);
                        //转换成c++字节序
                        msg_id=ntohl(msg_id);
                        msg_len = ntohl(msg_len);
                        //判断消息是否合法
                        if(msg_len>MAX_LEN||msg_len<0)
                        {
                            //回发送提示（入队）
                            Write(0,"recive message , but Length is error");
                            //停止接收后续数据，发完队列中的提示后由DoSend自动关闭连接
                            stop=true;
                            return;
                        }
                        //打印提示
                        std::cout<<"recviced message ,ID: "<<msg_id<<" and len : "<<msg_len<<std::endl;
                        //调用读信息函数
                        ReadBody(msg_id,msg_len);
                    }
        );
    }

        //异步读消息体
    void Session::ReadBody(int msg_id,int msg_len)
    {
        //创建读缓存
        recv_node=std::make_shared<RecvNode>(msg_len,msg_id);
        //清空创建缓存
        recv_node->Clear();
        auto self=shared_from_this();
        boost::asio::async_read(
            sock,boost::asio::buffer(recv_node->GetBuf(),recv_node->GetTotalLen()),
            [this,self,msg_id](const boost::system::error_code& ec,
                                std::size_t)
                                {
                                //判断是否有错误
                                if(ec){
                                //有，输出错误并返回
                                std::cout<<"error in run , value is : "<< ec.message()<<std::endl;
                                    return;
                                    }
                                //设置读的长度
                                recv_node->SetCurLen(recv_node->GetTotalLen());
                                //获得消息
                                std::string msg(recv_node->GetBuf(),recv_node->GetCurLen());
                                std::cout<<"recivced id : "<<msg_id<<" message is :"<<msg<<std::endl;
                                //调用工作函数
                                Work(msg_id,msg);
                                if(stop){
                                    //提示仍收到消息
                                    std::cout<<"closing,but recive message"<<std::endl;
                                    //回发送提示（入队，发送完毕后再关闭）
                                    Write(0,"server is closing,please not connect");
                                    return;
                                }
                                else{
                                    ReadHead();
                                }
                            }
                        );
    }

    //工作函数
    void Session::Work(int msg_id,std::string& msg)
    {
        std::string str="recivced message";
        Write(msg_id,str);
    }

        //异步写函数（将消息压入发送队列）
    void Session::Write(int msg_id, const std::string& msg)
    {
        //计算发送缓存大小
        int total_len=HEADER_LENGTH+static_cast<int>(msg.size());
        //创建发送缓存
        auto node=std::make_shared<SendNode>(total_len,msg_id);
        //获取创建的缓存指针
        char* buf =node->GetBuf();
        //获取消息体长度
        int value_len=static_cast<int>(msg.size());
        //改成网络字节序
        int32_t net_msg_id = htonl(msg_id);
        int32_t net_msg_len = htonl(value_len);
        //将msg_id拷贝进缓存
        std::memcpy(buf,&net_msg_id,HEAD_ID_LEN);
        //将msg_len拷贝进缓存
        std::memcpy(buf+HEAD_ID_LEN,&net_msg_len,HEAD_LEN);
        //将消息拷贝进缓存
        std::memcpy(buf+HEADER_LENGTH,msg.data(),value_len);
        //设置发送体的长度
        node->SetCurLen(total_len);
        //压入发送队列
        send_queue.push_back(node);
        //如果当前没有正在发送，则启动发送
        if(!is_sending){
            DoSend();
        }
    }

    //执行发送队列中的消息
    void Session::DoSend()
    {
    //队列为空则结束发送
        if(send_queue.empty()){
        is_sending=false;
        //若处于停止状态且队列已清空（此分支即队列为空），关闭连接
        if(stop && send_queue.empty()){
            boost::system::error_code ec;
            sock.close(ec);
        }
        return;
    }
        //标记正在发送
        is_sending=true;
        //取队头节点作为当前发送节点
        send_node=send_queue.front();
        //保活
        auto self=shared_from_this();
        //异步发送
        boost::asio::async_write(
            sock,boost::asio::buffer(send_node->GetBuf(),send_node->GetTotalLen()),
            [this,self](const boost::system::error_code& ec,std::size_t)
            {
                    if(ec){
                    std::cout<<"send error : "<<ec.message()<<std::endl;
                    //清空发送队列并复位发送状态
                    send_queue.clear();
                    is_sending=false;
                    //发送失败，关闭连接
                    boost::system::error_code close_ec;
                    sock.close(close_ec);
                    return;
                }
                std::cout<<"sended message"<<std::endl;
                //弹出已发送的队头节点
                send_queue.pop_front();
                //若停止状态且队列已空，关闭连接
                if(stop && send_queue.empty()){
                    boost::system::error_code ec;
                    sock.close(ec);
                    return;
                }
                //继续发送队列中的下一条消息
                DoSend();
            }
        );
    }

        //初始化Server
    Server::Server(boost::asio::io_context& _ioc,unsigned short port)
    :ioc(_ioc),acceptor(ioc,boost::asio::ip::tcp::endpoint(boost::asio::ip::tcp::v4(),port)){
        To_Session();
    }

        //Server停止函数（线程安全：可在任意线程调用，内部会投递到io_context线程执行）
    void Server::Stop(){
        //投递关闭操作到io_context线程，避免跨线程访问asio对象
        boost::asio::post(ioc,[this](){
            boost::system::error_code ec;
            acceptor.close(ec);

            //遍历并通知所有管理的会话
            for(auto& session:sessions)
            {
                session->Stop();
            }
            //清理所有的会话
            sessions.clear();
        });
    }

    //为连接创建Session
    void Server::To_Session(){
        //异步读
        acceptor.async_accept(
            [this](const boost::system::error_code& ec, boost::asio::ip::tcp::socket socket) {
            if (!ec) {
                std::cout << "conncet client"<<std::endl;
                auto session = std::make_shared<Session>(ioc,std::move(socket));
                sessions.push_back(session);
                session->Start();
                //继续接受下一个连接
                To_Session();
                }
            });
    }

    void start(unsigned short port){
        boost::asio::io_context io;
        Server server(io, port);
        boost::asio::signal_set sinals(io,SIGINT,SIGTERM);
        sinals.async_wait(
            [&](const boost::system::error_code& ec,int signal_number)
            {
                if(!ec){
                    std::cout<<std::endl<<"reciv close key :"<<signal_number<<", closing! worked after close!"<<std::endl;
                    //停服任务已投递到io_context线程执行
                    server.Stop();
                    //注意：不能在此处调用io.stop()，
                    //否则server.Stop()里post的任务来不及执行，io.run()会立即返回
                    //正确做法：让io.run()在处理完所有投递任务后自然退出
                }
            }
        );
        std::cout<<"listend"<<port<<std::endl;
        io.run();
    }
}
```
